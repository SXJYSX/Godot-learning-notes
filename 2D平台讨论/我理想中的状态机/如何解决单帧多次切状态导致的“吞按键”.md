按键被吞的根源在于：**玩家按键的瞬间只有 1 帧，而状态切换存在先后顺序。如果在不可切状态的那一帧按下，按键信号就会被漏掉。**

结合我们之前确定的 **物理优先级顺序（`Brain` $\rightarrow$ `StateMachine` $\rightarrow$ `Actor`）**，工业级的解决方案是 **“输入缓冲（Input Buffer）” + “单帧优先级裁决池”**。

#### 完整的 1 帧物理流水线（以“攻击结束前夕按跳跃”为例）：

假设在 60 FPS 的同一帧（第 N 帧）里，各个脚本按顺序执行：

Plaintext

```
============================== 第 N 帧物理循环 ==============================

[1. GoblinBrain / PlayerBrain (物理优先级 process_physics_priority = -10)]
  └─ 玩家在攻击收招前夕按下了跳跃。
  └─ Brain 启动输入缓冲：jump_buffer_timer = 0.15 秒 
     (按键意图被“锁存”在 Brain 变量中，绝不会当帧消失！)

[2. StateMachine & 当前 State (物理优先级 process_physics_priority = 0)]
  └─ 当前激活的是 AttackState，执行其 physics_update(delta)：
     ├─ 检查 A：攻击动画刚好在这一帧播放完毕。
     │   └─ 发信号: transition_requested.emit("IdleState", 0)  <-- 低优先级
     │
     └─ 检查 B：读取到 brain.has_jump_request() 为真。
         └─ 发信号: transition_requested.emit("JumpState", 100) <-- 高优先级

  └─ StateMachine 在本帧逻辑末尾执行 _resolve_pending_transitions()：
     ├─ 检查本帧收集到的申请池：[IdleState(0), JumpState(100)]
     ├─ 按 priority 从大到小排序，最高者胜出：JumpState(100)
     └─ 清空申请池，安全执行切换：AttackState.exit() -> JumpState.enter()

[3. Actor / Enemy (物理优先级 process_physics_priority = 10)]
  └─ JumpState 在 enter() 中赋予 actor.velocity.y = -JUMP_VELOCITY
  └─ 根节点执行 move_and_slide()，角色在这一帧顺利升空！
```

### 代码落地实现

#### 1. `Brain.gd` 中的输入缓冲（解决“按键生命周期过短”）

GDScript

```
# CharacterBrain.gd (大脑基类)
class_name CharacterBrain extends Node

var move_dir: Vector2 = Vector2.ZERO
var jump_buffer_timer: float = 0.0
const JUMP_BUFFER_TIME: float = 0.15 # 允许提前 0.15 秒预输入跳跃

func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed("jump"):
		jump_buffer_timer = JUMP_BUFFER_TIME # 锁存按键意图

func _physics_process(delta: float) -> void:
	if jump_buffer_timer > 0:
		jump_buffer_timer -= delta

# 提供给 State 用的读取接口
func has_jump_request() -> bool:
	return jump_buffer_timer > 0

# 消费型读取：读取后清空，防止连续跳跃
func consume_jump_request() -> bool:
	if jump_buffer_timer > 0:
		jump_buffer_timer = 0.0
		return true
	return false
```

#### 2. `StateMachine.gd` 中的优先级裁决池（解决“单帧多状态竞争”）

GDScript

```
# StateMachine.gd
class_name StateMachine extends Node

var current_state: State
# 存放本帧收到的所有切换申请
var pending_transitions: Array[Dictionary] = [] 

func _ready() -> void:
	for child in get_children():
		if child is State:
			child.transition_requested.connect(_on_transition_requested)

# 1. 收到信号时不立刻切，先存入申请池
func _on_transition_requested(target_name: String, priority: int) -> void:
	pending_transitions.append({
		"target": target_name,
		"priority": priority
	})

func physics_update(delta: float) -> void:
	if current_state:
		current_state.physics_update(delta)
	
	# 2. 在 StateMachine 更新的最后，集中裁决最高优先级的请求
	_resolve_pending_transitions()

func _resolve_pending_transitions() -> void:
	if pending_transitions.is_empty():
		return
		
	# 按优先级从大到小排序
	pending_transitions.sort_custom(func(a, b): return a["priority"] > b["priority"])
	
	# 取出优先级最高的申请
	var best_request = pending_transitions[0]
	pending_transitions.clear() # 清空池子，防止影响下一帧
	
	# 执行实际的状态切换
	_change_state(best_request["target"])

func _change_state(target_name: String) -> void:
	var new_state = get_node_or_null(target_name) as State
	if not new_state or new_state == current_state:
		return
		
	if current_state:
		current_state.exit()
		
	current_state = new_state
	current_state.enter()
```
